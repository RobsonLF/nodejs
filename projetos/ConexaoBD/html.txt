        <p class="intro">Insira, atualize ou exclua registros salvos no banco de dados.</p>

        <section class="panel" aria-labelledby="form-title">
            <h2 id="form-title">Novo usuário</h2>
            <form id="user-form">
                <div class="fields">
                    <label for="nome">

                    </label>

                        <input id="sobrenome" name="sobrenome" maxlength="100" autocomplete="family-name" required>

                    <button id="cancel-button" class="secondary" type="button" hidden>Cancelar edição</button>
                </div>
            </form>
            <p id="status" class="status" role="status" aria-live="polite"></p>
        </section>

    <script>
        const form = document.querySelector('#user-form');
        const nameInput = document.querySelector('#nome');
        const surnameInput = document.querySelector('#sobrenome');
        const list = document.querySelector('#user-list');
        const saveButton = document.querySelector('#save-button');
        const deleteButton = document.querySelector('#delete-button');
        const cancelButton = document.querySelector('#cancel-button');
        const formTitle = document.querySelector('#form-title');
        const statusMessage = document.querySelector('#status');
        let selectedId = null;

        function showStatus(message, isError = false) {
            statusMessage.textContent = message;
            statusMessage.classList.toggle('error', isError);
        }

        async function apiRequest(url, options = {}) {
            const response = await fetch(url, {
                ...options,
                headers: { 'Content-Type': 'application/json', ...options.headers },
            });
            const data = await response.json();
            if (!response.ok) {
                throw new Error(data.erro || 'Não foi possível concluir a operação.');
            }
            return data;
        }

        async function loadUsers() {
            const users = await apiRequest('/api/usuarios');
            list.replaceChildren();

            if (users.length === 0) {
                const row = document.createElement('tr');
                const cell = document.createElement('td');
                cell.colSpan = 4;
                cell.className = 'empty';
                cell.textContent = 'Nenhum usuário cadastrado.';
                row.append(cell);
                list.append(row);
                return;
            }

            for (const user of users) {
                const row = document.createElement('tr');
                for (const value of [user.id, user.nome, user.sobrenome]) {
                    const cell = document.createElement('td');
                    cell.textContent = value;
                    row.append(cell);
                }

                const actions = document.createElement('td');
                actions.className = 'actions-cell';

                const edit = document.createElement('button');
                edit.type = 'button';
                edit.className = 'secondary';
                edit.textContent = 'Editar';
                edit.addEventListener('click', () => beginEdit(user));

                const remove = document.createElement('button');
                remove.type = 'button';
                remove.className = 'danger';
                remove.textContent = 'Excluir';
                remove.addEventListener('click', () => removeUser(user.id));

                actions.append(edit, remove);
                row.append(actions);
                list.append(row);
            }
        }

        function beginEdit(user) {
            selectedId = user.id;
            nameInput.value = user.nome;
            surnameInput.value = user.sobrenome;
            formTitle.textContent = `Editando usuário #${user.id}`;
            saveButton.textContent = 'Salvar alterações';
            deleteButton.disabled = false;
            cancelButton.hidden = false;
            nameInput.focus();
            showStatus(`Usuário #${user.id} selecionado para edição.`);
        }

        function resetForm() {
            selectedId = null;
            form.reset();
            formTitle.textContent = 'Novo usuário';
            saveButton.textContent = 'Inserir usuário';
            deleteButton.disabled = true;
            cancelButton.hidden = true;
        }

        async function removeUser(id) {
            if (!window.confirm(`Deseja realmente excluir o usuário #${id}?`)) return;

            try {
                await apiRequest(`/api/usuarios/${id}`, { method: 'DELETE' });
                if (selectedId === id) resetForm();
                await loadUsers();
                showStatus(`Usuário #${id} excluído com sucesso.`);
            } catch (error) {
                showStatus(error.message, true);
            }
        }

        form.addEventListener('submit', async (event) => {
            event.preventDefault();
            const user = { nome: nameInput.value, sobrenome: surnameInput.value };
            const editing = selectedId !== null;

            try {
                await apiRequest(
                    editing ? `/api/usuarios/${selectedId}` : '/api/usuarios',
                    { method: editing ? 'PUT' : 'POST', body: JSON.stringify(user) },
                );
                resetForm();
                await loadUsers();
                showStatus(editing ? 'Usuário atualizado com sucesso.' : 'Usuário inserido com sucesso.');
            } catch (error) {
                showStatus(error.message, true);
            }
        });

        deleteButton.addEventListener('click', () => {
            if (selectedId !== null) removeUser(selectedId);
        });

        cancelButton.addEventListener('click', () => {
            resetForm();
            showStatus('Edição cancelada.');
        });

        loadUsers().catch((error) => showStatus(error.message, true));
    </script>
